% Template B: standard cross-ref-to-IDN-canonical notes.
% Args: #1 = country code (IDN|CHN|TZA), #2 = balance (unb|bal)
% Used by all main GRC tables EXCEPT the IDN cons/urban/unb canonical
% (which uses Template A) and EXCEPT the income tables (which use C).
\NewDocumentCommand{\GRCnotesxref}{m m}{%
  \csname datasource#1#2\endcsname{} \refdatasec{} \refIDNcanon. \seclusterind{} \sigstars
}

% Template A: IDN cons/urban/unb canonical (long-form variable explanations).
% No args; this is used in exactly ONE place. The "extrapolated returns"
% paragraph and column-by-column explanation live here.
\newcommand{\GRCnotesIDNcanonical}{%
  \datasourceIDNunb{} \refdatasec.
  The dependent variable is the log of total consumption per capita.
  Urban is an indicator equal to one for individuals who report living in a city or town, as opposed to a village.
  Individuals are assigned to trajectories based on their location history across the survey waves.
  This table reports the extrapolated returns to migrating to an urban location for individuals who are never observed in an urban location in the data.
  Columns~(2) to~(5) include time (survey wave) fixed effects, column~(3) adds a female indicator, column~(4) adds age squared, and column~(5) adds education (years of schooling, maximum across periods) and its square.
  \seclusterind{} \sigstars
}

% \GRCtable: main 5_GrRC.do tables (5 columns, c0/ct/c1/c2/ca).
%
%   Args: #1 = country code (IDN|CHN|TZA)
%         #2 = depvar (consumption|income)
%         #3 = choice (urban|nonag)
%         #4 = balance (unb|bal)
%         #5 = OPTIONAL notes-prefix override, default empty.
%               When set, the entire notes block is REPLACED by this string
%               (use the canonical macro: e.g. [\GRCnotesIDNcanonical] or
%                [\GRCnotesIncomeShared]).
%               When unset, defaults to Template B (\GRCnotesxref{#1}{#4}).
%
% Caption template:  "Restricted GRC Estimates of the Returns to <Treatment>
%   on log <Depvar> {in <CountryName> | , <CountryName>}[, Balanced Panel]"
% where the connector (in / ,) depends on depvar (in for cons, , for income).
% Treatment derived from #3 (urban→\treatmentUrban, nonag→\treatmentNonag).
% Depvar from #2 (consumption→Consumption, income→Income).
% Balance modifier appended only if #4 == bal.
\NewDocumentCommand{\GRCtable}{m m m m O{}}{%
  % Resolve the variable parts.
  \def\GRCthistreatment{\csname treatment\GRCtreatmentcase{#3}\endcsname}%
  \def\GRCthisdepvar{\csname depvar\GRCdepvarcase{#2}\endcsname}%
  \def\GRCthiscountry{\csname countryname#1\endcsname}%
  \def\GRCthisconnector{\GRCdepvarconnector{#2}}%
  \def\GRCthisbalmod{\GRCbalmodifier{#4}}%
  % Resolve notes: if #5 nonempty use it; else default to Template B.
  \def\GRCthisnotes{#5}%
  \ifx\GRCthisnotes\@empty
    \def\GRCthisnotes{\GRCnotesxref{#1}{#4}}%
  \fi
  \begin{table}[htbp]\centering
  \caption{Restricted GRC Estimates of the Returns to \GRCthistreatment{} on log \GRCthisdepvar{} \GRCthisconnector{} \GRCthiscountry\GRCthisbalmod}%
  \label{tab:GRC_#1_#2_#3_#4}
  \begin{threeparttable}
  \input{tables/GRC_#1_#2_#3_#4}%
  \begin{tablenotes}[flushleft]\footnotesize
    \item \GRCthisnotes
  \end{tablenotes}
  \end{threeparttable}
  \end{table}%
}

% Helpers: case-translation from arg lowercase to macro suffix CamelCase.
\newcommand{\GRCtreatmentcase}[1]{%
  \ifstrequal{#1}{urban}{Urban}{%
  \ifstrequal{#1}{nonag}{Nonag}{%
  ERROR-bad-treatment}}%
}
\newcommand{\GRCdepvarcase}[1]{%
  \ifstrequal{#1}{consumption}{Consumption}{%
  \ifstrequal{#1}{income}{Income}{%
  ERROR-bad-depvar}}%
}
\newcommand{\GRCdepvarconnector}[1]{%
  \ifstrequal{#1}{consumption}{in}{%
  \ifstrequal{#1}{income}{,}{%
  ERROR-bad-depvar}}%
}
\newcommand{\GRCbalmodifier}[1]{%
  \ifstrequal{#1}{bal}{, Balanced Panel}{}%
}
% NOTE: \ifstrequal comes from etoolbox; if not loaded, add \usepackage{etoolbox}.

% ---------------------------------------------------------------------
% \GRCexptable: experience and birth family tables (4 cols).
%   Args: #1=country, #2=depvar, #3=choice, #4=balance, #5=variant
%         where variant in {exp, exp_max, exp_sh, exp_m_sh, birth}
%
% Caption: "Restricted GRC Estimates of the Returns to <Treatment>
%   on log <Depvar> {in|,} <CountryName>[, Balanced Panel], <VariantName> Controls"
%
% Notes: cross-references the corresponding NON-variant table
%   (e.g. variant=exp on IDN cons urban unb cross-refs to
%   Table \ref{tab:GRC_IDN_consumption_urban_unb}).
% ---------------------------------------------------------------------
\NewDocumentCommand{\GRCexptable}{m m m m m}{%
  \def\GRCthistreatment{\csname treatment\GRCtreatmentcase{#3}\endcsname}%
  \def\GRCthisdepvar{\csname depvar\GRCdepvarcase{#2}\endcsname}%
  \def\GRCthiscountry{\csname countryname#1\endcsname}%
  \def\GRCthisconnector{\GRCdepvarconnector{#2}}%
  \def\GRCthisbalmod{\GRCbalmodifier{#4}}%
  \def\GRCthisvariant{\GRCvariantname{#5}}%
  \begin{table}[htbp]\centering
  \caption{Restricted GRC Estimates of the Returns to \GRCthistreatment{} on log \GRCthisdepvar{} \GRCthisconnector{} \GRCthiscountry\GRCthisbalmod, \GRCthisvariant{} Controls}%
  \label{tab:GRC_#1_#2_#3_#4_#5}
  \begin{threeparttable}
  \input{tables/GRC_#1_#2_#3_#4_#5}%
  \begin{tablenotes}[flushleft]\footnotesize
    \item \csname datasource#1#4\endcsname{} \refdatasec{} \refIDNcanon. \seclusterind{} \sigstars
  \end{tablenotes}
  \end{threeparttable}
  \end{table}%
}

% Variant token to caption-prose mapping.
\newcommand{\GRCvariantname}[1]{%
  \ifstrequal{#1}{exp}{Experience}{%
  \ifstrequal{#1}{exp_max}{Max Experience}{%
  \ifstrequal{#1}{exp_sh}{Experience Share}{%
  \ifstrequal{#1}{exp_m_sh}{Max Experience Share}{%
  \ifstrequal{#1}{birth}{Urban Birth}{%
  ERROR-bad-variant}}}}}%
}
