% ===============================================================
\begin{frame}{The pipeline had drifted into twenty-two .do files we could not audit}

\begin{itemize}\setlength\itemsep{4pt}
    \item Twenty-two numbered .do files; experience and birth families spread across six of them
    \item A second Dropbox folder, \code{ReplicationPackage6 - real values/}, kept manually in sync with the nominal-values one
    \item Ster filenames colliding across .do files, with several too long for Stata's internal name limit
\end{itemize}

\bigskip
None of this changes the published numbers.
All of it makes the pipeline harder to maintain.

\end{frame}

% ===============================================================
\begin{frame}{Experience and birth families: one program, one dispatcher}

The five experience .do files plus \code{15\_GrRC\_birth.do} all did the same thing: GRC regression with one extra regressor.

\medskip
\textcolor{cktblue}{What changed}
\begin{itemize}\setlength\itemsep{2pt}
    \item Shared regression logic: \code{run\_grc\_with\_extra\_regressor} in \code{0\_programs.do}
    \item Per-cell calls: \code{GRC\_extras.do}, one cell per (country, spec3, regressor)
    \item Tables: \code{extras\_tex\_table} reads sters and emits the family .tex files
\end{itemize}

\medskip
\textcolor{cktblue}{To add a new family member}
\begin{itemize}\setlength\itemsep{2pt}
    \item One cell in \code{GRC\_extras.do}
    \item One arm in \code{run\_grc\_with\_extra\_regressor}
\end{itemize}

\end{frame}

% ===============================================================
\begin{frame}{One table-builder for all GRC tables}

We had multiple table-building programs that were 95\% the same code:

\medskip
\begin{itemize}\setlength\itemsep{2pt}
    \item \code{grc\_tex\_table\_trend}
    \item \code{grc\_tex\_table\_trend\_exp}
    \item \code{grc\_tex\_table\_trend\_birth}
    \item \code{grc\_tex\_table\_trend\_hukou}
\end{itemize}

\medskip
Now: one \code{grc\_tex\_table\_trend}, four call signatures.

\bigskip
\textcolor{cktblue}{To change how a GRC table renders}---column widths, indicator-row order, postfoot string, decimal precision---edit \code{grc\_tex\_table\_trend} in \code{0\_programs.do}.

\end{frame}

% ===============================================================
\begin{frame}{One switch handles nominal and real values}

\textcolor{cktgray}{Before}
\begin{itemize}\setlength\itemsep{2pt}
    \item Two parallel Dropbox folders
    \item Every edit propagated by hand
\end{itemize}

\medskip
\textcolor{cktblue}{Now}
\begin{itemize}\setlength\itemsep{2pt}
    \item One global, \code{values(nominal|real)}, in \code{0\_path\_config.do}
    \item \code{nominal}: filenames as before
    \item \code{real}: appends \code{\_r} suffix; both modes coexist in one \code{output/} tree
\end{itemize}

\medskip
\textcolor{cktblue}{To switch modes}: edit \code{global values} in \code{0\_path\_config.do}.

\end{frame}

% ===============================================================
\begin{frame}{How ster filenames are built}

\textcolor{cktblue}{Pattern}: \code{grc\_<country>\_<spec3>\_<covs2>[\_<sfx1>].ster}

\bigskip
\begin{tabular}{l l}
\code{country} & \code{IDN}, \code{CHN}, \code{TZA}, or \code{CHN\_hukou\_<subgroup>} \\
\code{spec3}   & \code{cuu}, \code{cub}, \code{iuu}, \code{cnu} \\
\code{covs2}   & \code{c0}, \code{ct}, \code{c1}, \code{c2}, \code{ca} \\
\code{sfx1}    & \code{n} (never), \code{a} (always), \code{d} (delta), \code{g} (average) \\
\end{tabular}

\bigskip
Every ster filename is built from these tokens.
To find the ster behind a particular cell, assemble the tokens.

\bigskip
Full convention in \code{RP7/scripts/STER\_NAMING.md}.

\end{frame}

% ===============================================================
\begin{frame}{Editing the GRC tables}

\textcolor{cktblue}{To change caption, label, or notes}
\begin{itemize}\setlength\itemsep{2pt}
    \item Edit the \code{\textbackslash GRCtable\{\}} and \code{\textbackslash GRChukoutable\{\}} macros in \code{preamble.tex} (the paper)
\end{itemize}

\medskip
\textcolor{cktblue}{To change which coefficients appear, label text, or formatting}
\begin{itemize}\setlength\itemsep{2pt}
    \item Per-cell options: \code{make\_tables.do}
    \item Underlying template: \code{grc\_tex\_table\_trend} in \code{0\_programs.do}
\end{itemize}

\medskip
\textcolor{cktblue}{Cosmetic refresh in this round}
\begin{itemize}\setlength\itemsep{2pt}
    \item Average $\Delta$ row reads $\bar{\Delta}$
    \item Spurious blank rows from \code{esttab} stripped after output
\end{itemize}

\end{frame}

% ===============================================================
\begin{frame}{What this buys us}

\begin{itemize}\setlength\itemsep{4pt}
    \item One source of truth for the regression code
    \item One source of truth for the table caption, label, and notes (in the paper preamble)
    \item Ster filenames you can read, type, and look up
    \item Same published numbers as before
\end{itemize}

\bigskip
The pipeline is now small enough to hold in one head.

\end{frame}
