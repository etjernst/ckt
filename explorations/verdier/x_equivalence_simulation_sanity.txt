   207 . * **********************************************************************
   208 . * gen_vfirst: earliest non-missing value of `vname' per pid
   209 . * **********************************************************************
   210 . * Used by initial_values_robust / run_grc_robust to build the
   211 . * time-invariant cluster index v_i (first-wave province/region) that
   212 . * VV (2020) Section F requires for the robust extrapolation.
   213 . * Unit-tested at explorations/verdier/3_test_gen_vfirst.do.
   214 . * The buggy bysort pid: egen min(cond(!missing(v), v, .)) returns the
   215 . * smallest numeric value of v per pid, NOT the first-wave value; the
   216 . * correct form below marks the FIRST non-missing row per pid sorted by
   217 . * year, then propagates its value within pid.
   218 . capture program drop gen_vfirst
      
   219 . program define gen_vfirst
        1.     syntax , vname(varname) genname(name)
        2.     capture drop `genname'
        3.     tempvar seq mark tmp
        4.     bysort pid (year): gen `seq' = sum(!missing(`vname'))
        5.     by pid: gen `mark' = (!missing(`vname')) & (`seq' == 1)
        6.     gen `tmp' = `vname' if `mark'
        7.     by pid: egen `genname' = max(`tmp')
        8.     lab var `genname' "First-wave non-missing `vname'"
        9. end

   809 . * **********************************************************************
   810 . * run_grc_robust_vv: VV-style robust extrapolation via cluster-demeaned
   811 . *                    instruments (no cluster-FE parameters)
   812 . * **********************************************************************
   813 . * Ports the core identification idea of Verdier (2020) Online Appendix
   814 . * Section F / Table1/Code/robust.do to CKT's trajectory-pooled GRC.
   815 . *
   816 . * Key difference vs run_grc_robust (single-step cluster-dummy version):
   817 . *   - run_grc_robust adds |V|-1 cluster*choice interactions as free
   818 . *     parameters (beta_dev_v). Asymptotically equivalent to VV's
   819 . *     approach, but in finite sample with few clusters the extra |V|-1
   820 . *     parameters make the GMM objective nearly flat in phi, giving
   821 . *     multiple local minima (confirmed on TZA; hung on CHN).
   822 . *   - run_grc_robust_vv keeps CKT's original parameters (phi, mu_d,
   823 . *     Delta_base, kappa, xb gammas) -- no cluster FEs -- and instead
   824 . *     replaces the switcher_s_choice instruments with their within-
   825 . *     cluster-demeaned residuals (regressed on i.vfirst among workers
   826 . *     in trajectory s). The within-cluster variation in treatment
   827 . *     identifies phi robustly to cluster-level confounders in the
   828 . *     outcome, matching VV's Prop. F identification assumption.
   829 . *
   830 . * This mirrors VV's Table1/Code/robust.do: he demeans hybrid_per on
   831 . * i.vil among switchers (line 12), uses the residual as the optimal
   832 . * instrument (line 14), and fits GMM with vce(cluster vil), winitial
   833 . * unadjusted independent, onestep (line 206). We follow the same
   834 . * pattern but adapted to CKT's trajectory structure.
   835 . *
   836 . * Aggregations for _never etc. follow VV Eq. F.3 conceptually -- for
   837 . * P1 we defer the full cluster-share-weighted aggregator to P2 and use
   838 . * the simple-spec extrapolation with VV-style phi.

  1038 . * ============================================================
  1039 . * DGP parameters
  1040 . * ============================================================
  1041 . local N            50000
      
  1042 . local T            4
      
  1043 . local V            25
      
  1044 . local phi_true     -0.7
      
  1045 . local sigma_theta  0.5
      
  1046 . local sigma_xi     0.3
      
  1047 . local sigma_u      0.3
      
  1048 . local sigma_beta   0.4
      
  1049 . local c1           1.0
      
  1050 . local c2           0.5
      
  1051 . local m_slope      0.1     // m_s = m_slope * (s - 8.5)
      
  1052 . local lambda_r2a   1.0     // for differential cluster assignment (R2a)
      
  1053 . local lambda_r2b   1.0     // for theta-tilt (R2b)
      
  1054 . local seed         20260425
      
  1055 . 
  1056 . set seed `seed'
      
  1057 . 
  1058 . * ============================================================
  1059 . * Generate synthetic panel under R1 (A3 holds, uniform cluster assignment)
  1060 . * ============================================================
  1061 . clear
      
  1062 . set obs `N'
      Number of observations (_N) was 0, now 50,000.
      
  1063 . gen long pid = _n
      
  1064 . 
  1065 . * Trajectory: uniform over 1..16. Trajectory 1 = never (RRRR), 16 = always (U
      > UUU).
  1066 . gen int trajectory = 1 + floor(16 * runiform())
      
  1067 . gen byte traj = trajectory   // alias for DGP formulas below
      
  1068 . 
  1069 . * Cluster: uniform 1..V
  1070 . gen byte vfirst_true = 1 + floor(`V' * runiform())
      
  1071 . 
  1072 . * Worker-level comparative advantage theta_i. Under R1: m_s depends only
  1073 . * on trajectory.
  1074 . gen double m_s = `m_slope' * (traj - 8.5)
      
  1075 . gen double theta_i = m_s + `sigma_theta' * rnormal()
      
  1076 . 
  1077 . * Cluster-specific intercept beta(v) = sigma_beta * h(v), where
  1078 . * h(v) = (v - (V+1)/2) / ((V-1)/2). For V=25, h(v) in [-1, 1].
  1079 . gen double h_v = (vfirst_true - (`V'+1)/2) / ((`V'-1)/2)
      
  1080 . gen double beta_v = `sigma_beta' * h_v
      
  1081 . 
  1082 . * Worker intercept: alpha_i = c1*theta_i + c2*eta_i. Non-degenerate.
  1083 . gen double eta_i = rnormal()
      
  1084 . gen double alpha_i = `c1' * theta_i + `c2' * eta_i
      
  1085 . 
  1086 . * Worker-level xi (deviation from LCA)
  1087 . gen double xi_i = `sigma_xi' * rnormal()
      
  1088 . 
  1089 . * Worker treatment effect: Delta_i = beta(v_i) + phi * theta_i + xi_i
  1090 . gen double Delta_i = beta_v + `phi_true' * theta_i + xi_i
      
  1091 . 
  1092 . * Expand to panel
  1093 . expand `T'
      (150,000 observations created)
      
  1094 . bysort pid: gen byte t = _n
      
  1095 . gen int period = t
      
  1096 . gen int year = 2000 + t   // arbitrary year for compatibility
      
  1097 . 
  1098 . * Treatment D_it = bit_(T-t+1) of (traj - 1), MSB to LSB
  1099 . gen byte D_it = mod(floor((traj - 1) / 2^(`T' - t)), 2)
      
  1100 . rename D_it choice
      
  1101 . 
  1102 . * Outcome
  1103 . gen double u_it = `sigma_u' * rnormal()
      
  1104 . gen double lndepvar = alpha_i + Delta_i * choice + u_it
      
  1105 . 
  1106 . * Covariates placeholder (setup_grc_estimation expects these)
  1107 . gen byte female = 0
      
  1108 . gen byte age = 40
      
  1109 . gen double age2 = age^2
