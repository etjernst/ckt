   207 . * **********************************************************************
   208 . * gen_vfirst: earliest non-missing value of `vname' per pid
   209 . * **********************************************************************
   210 . * Used by initial_values_robust / run_grc_robust to build the
   211 . * time-invariant cluster index v_i (first-wave province/region) that
   212 . * VV (2020) Section F requires for the robust extrapolation.
   213 . * Unit-tested at explorations/verdier/3_test_gen_vfirst.do.
   214 . * The buggy bysort pid: egen min(cond(!missing(v), v, .)) returns the
   215 . * smallest numeric value of v per pid, NOT the first-wave value; the
   216 . * correct form below marks the FIRST non-missing row per pid sorted by
   217 . * year, then propagates its value within pid.
   218 . capture program drop gen_vfirst
      
   219 . program define gen_vfirst
        1.     syntax , vname(varname) genname(name)
        2.     capture drop `genname'
        3.     tempvar seq mark tmp
        4.     bysort pid (year): gen `seq' = sum(!missing(`vname'))
        5.     by pid: gen `mark' = (!missing(`vname')) & (`seq' == 1)
        6.     gen `tmp' = `vname' if `mark'
        7.     by pid: egen `genname' = max(`tmp')
        8.     lab var `genname' "First-wave non-missing `vname'"
        9. end

       89.     local converged_str = cond(e(converged)==1, "Y", "N")

       91. 
   774 .     * Cluster diagnostics stored on the estimate for table builders

       92.     estadd scalar V_ge10sw   = `nclust_ge10', replace : `estname'
       93.     estadd scalar V_supp     = `V_supp',      replace : `estname'
       94.

   777 .     estimates save "$dir/output/`estname'", replace
       95. 
   778 .     * ----------------------------------------------------------------
   779 .     * Delta_never: cluster-share-weighted aggregate over V_supp
   780 .     *   Delta_never = Delta_base + sum_{v != v_base, v in V_supp} w_v * beta_
      > dev_v
   781 .     *                            + phi * (mu:never - mu:switcher_base)
   782 .     *   w_v = n^N_v / sum_{v' in V_supp} n^N_{v'}
   783 .     * (Delta_base absorbs the baseline-cluster weight since weights sum to 1
   784 .     *  and beta_dev_{v_base} is 0 by the drop-a-dummy convention.)
   785 .     * ----------------------------------------------------------------
   786 .     estimates restore `estname'
       96. 
   787 .     * Recompute support + per-cluster never counts (vchoice_list-indexed only
      > )
   788 .     preserve
       97.         qui duplicates drop vfirst, force
       98.         qui keep if `n_sw_v' > 0 & `n_nev_v' > 0

      100.         local nev_supp_sum = r(sum)
      101. 
   789 .         * Build weighted sum of beta_dev terms

      102.         if `nev_supp_sum' > 0 & "`vchoice_list'" != "" {
      103.             forvalues i = 1/`=_N' {
      104.                 local v_i = vfirst[`i']
      105.                 local n_nev_i = `n_nev_v'[`i']
      106.                 if `v_i' != `v_base' {
      107.                     local w_i = `n_nev_i' / `nev_supp_sum'
      108.                     local beta_agg_expr "`beta_agg_expr' + `w_i' * _b[beta
      > _dev:vchoice_`v_i']"
      109.                 }
      110.             }
      111.         }
      112.     restore
      113. 
   791 .     di as text "run_grc_robust: Delta_never aggregate expression:"
      114.     di as text "  `beta_agg_expr' + phi * (mu_never - mu_base)"
      115. 
   792 .     nlcom (Delta_never: `beta_agg_expr' + _b[phi:_cons] * ///
      >             (_b[mu:never] - _b[mu:switcher_`base'])), post
      116.     estadd scalar V_never_supp = `V_supp', replace
      117.     estimates save "$dir/output/`estname'_never", replace
      118. 
   793 .     * ----------------------------------------------------------------
   794 .     * Delta_always (baseline-cluster beta; proper aggregation in P2)
   795 .     * ----------------------------------------------------------------
   796 .     estimates restore `estname'
      119.     nlcom (Delta_always: _b[Delta_base:_cons] + (_b[phi:_cons] * ///
      >             (_b[kappa:_cons] - _b[mu:switcher_`base']))), post
      120.     estimates save "$dir/output/`estname'_always", replace
      121. 
   797 .     * ----------------------------------------------------------------
   798 .     * Per-switcher Deltas (baseline-cluster beta) + joint test
   799 .     * ----------------------------------------------------------------
   800 .     estimates restore `estname'
      122.     local nlcom_expr ""
      123.     foreach s of numlist $switchers {
      124.         local nlcom_expr "`nlcom_expr' (Delta_`s': _b[Delta_base:_cons] + 
      > (_b[phi:_cons] * (_b[mu:switcher_`s'] - _b[mu:switcher_`base'])))"
      125.     }
      126.     nlcom `nlcom_expr', post
      127. 
   801 .     local d_test ""
      128.     local s0 : word 1 of $switchers
      129.     local d_test "Delta_`s0'"
      130.     foreach s of numlist $switchers {
      131.         if `s' != `s0' {
      132.             local d_test "`d_test' = Delta_`s'"
      133.         }
      134.     }
      135.     test `d_test'
      136.     estadd scalar joint_chi2 = r(chi2), replace
      137.     estadd scalar joint_p    = r(p),    replace
      138.     estimates save "$dir/output/`estname'_delta", replace
      139. 
   802 .     * ----------------------------------------------------------------
   803 .     * Delta_avg: trajectory-share-weighted average across switchers
   804 .     * (baseline-cluster beta; mirror run_grc structure)
   805 .     * ----------------------------------------------------------------
   806 .     local first_loop = 1
      140.     local Delta_avg_nlcom ""
      141.     foreach s of numlist $switchers {
      142.         estimates restore `estname'
      143.         sum 1.switcher_`s' if e(sample)
      144.         local num_`s' = r(mean)

      > lta_base:_cons] + (_b[phi:_cons] * (_b[mu:switcher_`s'] - _b[mu:switcher_`bas
      > e']))))"
      147.         }
      148.         else if `first_loop' == 1 {
      149.             local Delta_avg_nlcom "(`num_`s'' * (_b[Delta_base:_cons] + (_
      > b[phi:_cons] * (_b[mu:switcher_`s'] - _b[mu:switcher_`base']))))"
      150.             local first_loop = 0
      151.         }
      152.     }
      153.     estimates restore `estname'
